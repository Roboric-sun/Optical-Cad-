cmake_minimum_required(VERSION 3.24)
project(OpticalCAD VERSION 0.8.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options("$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
option(OPTICS_BUILD_GUI "Build the Qt desktop application" ON)
option(OPTICS_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
add_library(optics_core src/core/material.cpp src/core/sequential.cpp src/core/analysis.cpp src/core/nonsequential.cpp src/core/optimization.cpp src/core/detector.cpp src/core/solves.cpp src/core/field.cpp)
target_include_directories(optics_core PUBLIC include)
if(MSVC)
  target_compile_options(optics_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(optics_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(OPTICS_SANITIZERS AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(optics_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
enable_testing()
add_executable(optics_tests tests/core_tests.cpp tests/optimization_tests.cpp tests/solve_tests.cpp tests/marginal_tests.cpp tests/field_tests.cpp tests/field_type_tests.cpp)
target_link_libraries(optics_tests PRIVATE optics_core)
add_test(NAME optics_physics COMMAND optics_tests)
add_executable(optics_cli src/cli.cpp)
target_link_libraries(optics_cli PRIVATE optics_core)
add_executable(optics_learning examples/learning/first_lens.cpp)
target_link_libraries(optics_learning PRIVATE optics_core)
if(OPTICS_BUILD_GUI)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Concurrent Test)
  set(CMAKE_AUTOMOC ON)
  qt_add_executable(optical_cad WIN32 MACOSX_BUNDLE src/app/main.cpp src/app/window.cpp src/app/office.cpp src/app/plots.cpp src/app/project.cpp src/app/geopter_import.cpp src/app/optimization_editor.cpp src/app/solve_editor.cpp src/app/learning_help.cpp src/app/field_editor.cpp)
  target_link_libraries(optical_cad PRIVATE optics_core Qt6::Widgets Qt6::Concurrent)
  target_include_directories(optical_cad PRIVATE src/app)
  set_target_properties(optical_cad PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Optical CAD" MACOSX_BUNDLE_GUI_IDENTIFIER "ru.opticalcad.desktop" MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION})
  qt_add_executable(optics_gui_tests tests/gui_tests.cpp tests/gui_improvement_tests.cpp tests/combo_tests.cpp tests/solve_gui_tests.cpp tests/marginal_gui_tests.cpp tests/field_gui_tests.cpp tests/field_type_gui_tests.cpp src/app/window.cpp src/app/office.cpp src/app/plots.cpp src/app/project.cpp src/app/geopter_import.cpp src/app/optimization_editor.cpp src/app/solve_editor.cpp src/app/learning_help.cpp src/app/field_editor.cpp)
  target_include_directories(optics_gui_tests PRIVATE src/app)
  target_link_libraries(optics_gui_tests PRIVATE optics_core Qt6::Widgets Qt6::Concurrent Qt6::Test)
  target_compile_definitions(optics_gui_tests PRIVATE OPTICS_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  foreach(learning_target optical_cad optics_gui_tests)
    qt_add_resources(${learning_target} learning_guide PREFIX "/learning"
      BASE "${CMAKE_CURRENT_SOURCE_DIR}/docs" FILES docs/LEARNING_CPP_RU.md)
  endforeach()
  add_test(NAME optics_gui COMMAND optics_gui_tests)
  set_tests_properties(optics_gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  if(WIN32)
    # The offscreen plugin uses FreeType rather than the native Windows font
    # database. Point it at system fonts for real glyphs and layout metrics.
    file(TO_CMAKE_PATH "$ENV{SystemRoot}/Fonts" OPTICS_TEST_FONT_DIR)
    set_property(TEST optics_gui APPEND PROPERTY ENVIRONMENT "QT_QPA_FONTDIR=${OPTICS_TEST_FONT_DIR}")
  endif()
  install(TARGETS optical_cad BUNDLE DESTINATION . RUNTIME DESTINATION bin)
  install(DIRECTORY examples DESTINATION .)
  install(FILES README.md DESTINATION .)
  install(FILES scripts/reference_doublet.py DESTINATION scripts)
  install(FILES docs/REQUIREMENTS.md docs/SOURCES.md docs/TESTING.md docs/UI_REFERENCE.md docs/CHANGELOG.md docs/GEOPTER_PARITY.md docs/LEARNING_CPP_RU.md docs/ROADMAP_RU.md docs/BENCHMARKS_RU.md docs/FIELDS_RU.md DESTINATION docs)
  if(APPLE)
    find_program(OPTICS_QMAKE NAMES qmake6 qmake HINTS "${Qt6_DIR}/../../../bin" REQUIRED)
    execute_process(COMMAND "${OPTICS_QMAKE}" -query QT_INSTALL_BINS OUTPUT_VARIABLE OPTICS_QT_BINS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    execute_process(COMMAND "${OPTICS_QMAKE}" -query QT_INSTALL_PLUGINS OUTPUT_VARIABLE OPTICS_QT_PLUGINS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    execute_process(COMMAND "${OPTICS_QMAKE}" -query QT_INSTALL_LIBS OUTPUT_VARIABLE OPTICS_QT_LIBS OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    find_program(OPTICS_MACDEPLOYQT NAMES macdeployqt6 macdeployqt HINTS "${OPTICS_QT_BINS}" REQUIRED)
    configure_file(cmake/deploy_macos.cmake.in deploy_macos_configured.cmake @ONLY)
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/deploy_macos.cmake" INPUT "${CMAKE_CURRENT_BINARY_DIR}/deploy_macos_configured.cmake")
    install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/deploy_macos.cmake")
  else()
    qt_generate_deploy_app_script(TARGET optical_cad OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
    install(SCRIPT ${deploy_script})
  endif()
endif()
